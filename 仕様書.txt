## 1. プロジェクト概要

### 1.1 ゲームコンセプト
```

- ジャンル: タクティカル・ローグライト・オートバトラー
- プレイ時間: 1ラン 30-60分

## 2. コアゲームループ

### 2.1 1ランの流れ
```

[開始]
↓
スターターデッキ選択（3体）
↓
ノードマップ表示（15ノード）
↓
ノード選択 → イベント/バトル実行
↓
報酬獲得 → パーティ強化
↓
（10-15ノード繰り返し）
↓
ボス戦
↓
[クリア or 全滅]
↓
報酬確定 → 図鑑解放 → 次のラン

```
### 2.2 バトルの流れ
```

[戦闘開始]
↓
配置フェーズ（30秒）

- 3×3グリッドに最大6体配置
- 各モンスターの行動ルール設定
  ↓
  実行フェーズ（オート）
- ターン制で自動実行
- 倍速可能（×1, ×2, ×5）
  ↓
  （オプション）タイムアウト介入
- 1戦につき1回のみ
- 配置変更 or 手動技発動
  ↓
  [勝利 or 敗北]

```

## 4. バトルシステム詳細

### 4.1 グリッド配置

```
3×3グリッド（プレイヤー側）

[0,0] [0,1] [0,2]  ← 前列（最初に狙われる）
[1,0] [1,1] [1,2]  ← 中列
[2,0] [2,1] [2,2]  ← 後列（守られる）

敵も同様に3×3グリッド配置

射程の概念:
- 射程1: 同じ列の最前列のみ攻撃可能
- 射程2: 同じ列の前列・中列攻撃可能
- 射程3: 全ての敵を攻撃可能
```

### 4.2 ターン処理

```javascript
// 1ターンの処理フロー
function executeTurn() {
  // 1. 行動順序決定（SPD順）
  const actionOrder = [...playerMonsters, ...enemies]
    .sort((a, b) => b.stats.spd - a.stats.spd);
  
  // 2. 順番に行動
  actionOrder.forEach(unit => {
    if (unit.currentHp <= 0) return; // 戦闘不能チェック
    
    // 行動決定
    const action = decideAction(unit);
    
    // 実行
    executeAction(unit, action);
    
    // 勝敗判定
    if (checkBattleEnd()) return;
  });
  
  // 3. ターン終了処理
  applyEndOfTurnEffects();
  turnCount++;
}

// 行動決定ロジック
function decideAction(unit) {
  // プレイヤーユニット: 設定されたbehaviorに従う
  if (unit.isPlayer) {
    const targets = findTargets(unit.behavior.priority);
    const skill = selectSkill(unit, targets);
    return { skill, targets };
  }
  
  // 敵: AI パターンに従う
  return enemyAI(unit);
}
```

### 4.3 ダメージ計算

```javascript
function calculateDamage(attacker, defender, skill) {
  // 基本ダメージ
  let damage = skill.power + attacker.stats.atk - defender.stats.def;
  
  // タイプ相性（簡易版）
  const typeBonus = getTypeMultiplier(attacker.type, defender.type);
  damage *= typeBonus;
  
  // クリティカル（10%）
  if (Math.random() < 0.1) {
    damage *= 1.5;
    showCritical();
  }
  
  // 最低ダメージ保証
  damage = Math.max(1, Math.floor(damage));
  
  return damage;
}

// タイプ相性
function getTypeMultiplier(attackType, defenseType) {
  const chart = {
    fire: { grass: 1.5, water: 0.5 },
    water: { fire: 1.5, grass: 0.5 },
    grass: { water: 1.5, fire: 0.5 },
    light: { dark: 1.5 },
    dark: { light: 1.5 }
  };
  
  return chart[attackType]?.[defenseType] || 1.0;
}
```

### 5.2 マップ生成ロジック

```javascript
function generateMap(difficulty) {
  const map = {
    layers: []
  };
  
  // Layer 0: スタート
  map.layers.push({
    nodes: [{ type: "start" }]
  });
  
  // Layer 1-2: 序盤（戦闘中心）
  for (let i = 1; i <= 2; i++) {
    map.layers.push({
      nodes: generateNodes(3, ["battle", "battle", "event"], difficulty)
    });
  }
  
  // Layer 3: 中盤（休息・商人出現）
  map.layers.push({
    nodes: generateNodes(3, ["rest", "shop", "battle"], difficulty)
  });
  
  // Layer 4: 後半（エリート出現）
  map.layers.push({
    nodes: generateNodes(3, ["battle", "elite", "event"], difficulty)
  });
  
  // Layer 5: ボス前（最終準備）
  map.layers.push({
    nodes: generateNodes(2, ["shop", "rest"], difficulty)
  });
  
  // Layer 6: ボス
  map.layers.push({
    nodes: [{ type: "boss" }]
  });
  
  // ノード間の接続を生成
  connectNodes(map);
  
  return map;
}
```

## 12. 敵AI設計

### 12.1 AI行動パターン

```javascript
const EnemyAI = {
  // 攻撃的AI
  aggressive: (enemy, playerParty) => {
    // 最もHPが低い敵を狙う
    const target = playerParty
      .filter(m => m.currentHp > 0)
      .sort((a, b) => a.currentHp - b.currentHp)[0];
    
    // 最も火力の高いスキルを使用
    const skill = enemy.skills
      .sort((a, b) => b.power - a.power)[0];
    
    return { target, skill };
  },
  
  // 防御的AI
  defensive: (enemy, playerParty) => {
    // 自分のHPが50%以下なら回復
    if (enemy.currentHp < enemy.stats.hp * 0.5) {
      const healSkill = enemy.skills.find(s => s.effect === "heal");
      if (healSkill) return { target: enemy, skill: healSkill };
    }
    
    // それ以外は最も攻撃力が高い敵を狙う
    const target = playerParty
      .filter(m => m.currentHp > 0)
      .sort((a, b) => b.stats.atk - a.stats.atk)[0];
    
    return { target, skill: enemy.skills[0] };
  },
  
  // バランス型AI
  balanced: (enemy, playerParty) => {
    // 状況判断
    const myHpPercent = enemy.currentHp / enemy.stats.hp;
    const enemyCount = playerParty.filter(m => m.currentHp > 0).length;
    
    // 複数の敵がいて範囲攻撃持ちなら範囲攻撃
    if (enemyCount >= 2) {
      const aoeSkill = enemy.skills.find(s => 
        s.target === "all" || s.target === "row"
      );
      if (aoeSkill) return { target: "all", skill: aoeSkill };
    }
    
    // HPが低ければ防御行動
    if (myHpPercent < 0.3) {
      const defenseSkill = enemy.skills.find(s => 
        s.effect === "buff_def" || s.effect === "heal"
      );
      if (defenseSkill) return { target: enemy, skill: defenseSkill };
    }
    
    // 通常は最も厄介な敵を狙う
    const target = playerParty
      .filter(m => m.currentHp > 0)
      .sort((a, b) => {
        const threatA = a.stats.atk * (a.currentHp / a.stats.hp);
        const threatB = b.stats.atk * (b.currentHp / b.stats.hp);
        return threatB - threatA;
      })[0];
    
    return { target, skill: enemy.skills[0] };
  }
};
```

```javascript
function equipItem(monster, item) {
  // 装備スロット確認
  const slot = item.slot;
  
  // 既存装備があれば外す
  if (monster.equipment[slot]) {
    removeEquipmentBonus(monster, monster.equipment[slot]);
  }
  
  // 新しい装備をセット
  monster.equipment[slot] = item;
  
  // ステータスボーナス適用
  if (item.stats_bonus) {
    Object.keys(item.stats_bonus).forEach(stat => {
      monster.stats[stat] += item.stats_bonus[stat];
    });
  }
  
  // 特殊効果登録
  if (item.special_effect) {
    monster.special_effects.push(item.special_effect);
  }
  
  // HP上限が増えた場合、現在HPも増やす
  if (item.stats_bonus.hp) {
    monster.currentHp += item.stats_bonus.hp;
  }
}

function removeEquipmentBonus(monster, item) {
  // ステータスボーナス削除
  if (item.stats_bonus) {
    Object.keys(item.stats_bonus).forEach(stat => {
      monster.stats[stat] -= item.stats_bonus[stat];
    });
  }
  
  // HP上限が減る場合、現在HPも調整
  if (item.stats_bonus.hp) {
    monster.currentHp = Math.min(
      monster.currentHp,
      monster.stats.hp - item.stats_bonus.hp
    );
  }
  
  // 特殊効果削除
  if (item.special_effect) {
    monster.special_effects = monster.special_effects.filter(
      e => e !== item.special_effect
    );
  }
}

function executeEvent(eventId) {
  const event = Events.find(e => e.id === eventId);
  
  showEventDialog({
    title: event.name,
    description: event.description,
    image: event.image || "default_event.png",
    choices: event.choices.map(choice => ({
      text: choice.text,
      cost: choice.cost,
      enabled: canAffordCost(choice.cost),
      onClick: () => resolveChoice(choice)
    }))
  });
}

function resolveChoice(choice) {
  // コスト支払い
  if (choice.cost) {
    if (!payCost(choice.cost)) {
      addLog("コストが足りない！");
      return;
    }
  }
  
  // 結果判定（確率に基づく）
  const roll = Math.random();
  let cumulative = 0;
  
  for (const outcome of choice.outcomes) {
    cumulative += outcome.probability;
    if (roll <= cumulative) {
      outcome.effect();
      showOutcomeAnimation(outcome.result);
      break;
    }
  }
  
  // イベント終了後、次のノードへ
  proceedToNextNode();
}

function canAffordCost(cost) {
  if (!cost) return true;
  
  if (cost.gold && player.gold < cost.gold) return false;
  if (cost.item && !hasItem(cost.item, cost.amount)) return false;
  if (cost.monster && playerParty.length <= cost.monster) return false;
  
  return true;
}

function payCost(cost) {
  if (!canAffordCost(cost)) return false;
  
  if (cost.gold) player.gold -= cost.gold;
  if (cost.item) removeItem(cost.item, cost.amount);
  
  return true;
}

function gainExperience(monster, exp) {
  monster.experience += exp;
  
  addLog(`${monster.name} は ${exp} の経験値を獲得！`);
  
  // レベルアップチェック
  let leveledUp = false;
  while (monster.experience >= getExpToLevel(monster.level + 1)) {

    levelUp(monster);

function levelUp(monster) {
  monster.level++;
  
  // ステータス成長（モンスターの種族値に基づく）
  const baseGrowth = getBaseGrowthRates(monster.id);
  const variance = 0.2; // ±20%のランダム性
  
  const growth = {
    hp: Math.floor(baseGrowth.hp * (1 + (Math.random() - 0.5) * variance)),
    atk: Math.floor(baseGrowth.atk * (1 + (Math.random() - 0.5) * variance)),
    def: Math.floor(baseGrowth.def * (1 + (Math.random() - 0.5) * variance)),
    spd: Math.floor(baseGrowth.spd * (1 + (Math.random() - 0.5) * variance))
  };
  
  // ステータス適用
  Object.keys(growth).forEach(stat => {
    monster.stats[stat] += growth[stat];
  });
  
  // 現在HPも増加分回復
  monster.currentHp += growth.hp;
  
  // レベルアップ演出
  showLevelUpAnimation(monster);
  addLog(`${monster.name} がレベル ${monster.level} になった！`, "important");
  
  // ステータス上昇表示
  addLog(
    `HP+${growth.hp} ATK+${growth.atk} DEF+${growth.def} SPD+${growth.spd}`
  );
  
  // 特定レベルでスキル習得チャンス
  if (monster.level === 5 || monster.level === 10 || monster.level === 15) {
    showSkillLearnDialog(monster);
  }
}

### 16.2 セーブ・ロード関数

```javascript
// セーブ
function saveGame() {
  const saveData = {
    version: GAME_VERSION,
    timestamp: Date.now(),
    player: getPlayerData(),
    codex: getCodexData(),
    achievements: getAchievementData(),
    unlocks: getUnlocksData(),
    current_run: getCurrentRunData(),
    settings: getSettings()
  };
  
  try {
    // localStorageに保存
    localStorage.setItem('vmo_tactics_save', JSON.stringify(saveData));
    
    // バックアップ作成
    const previousSave = localStorage.getItem('vmo_tactics_save');
    if (previousSave) {
      localStorage.setItem('vmo_tactics_backup', previousSave);
    }
    
    showNotification("セーブしました", "success");
    return true;
  } catch (error) {
    console.error("Save failed:", error);
    showNotification("セーブに失敗しました", "error");
    return false;
  }
}

// ロード
function loadGame() {
  try {
    const saveDataStr = localStorage.getItem('vmo_tactics_save');
    
    if (!saveDataStr) {
      return initNewGame();
    }
    
    const saveData = JSON.parse(saveDataStr);
    
    // バージョンチェック
    if (saveData.version !== GAME_VERSION) {
      const migrated = migrateOldSave(saveData);
      if (!migrated) {
        throw new Error("Migration failed");
      }
      return migrated;
    }
    
    // データ復元
    restorePlayerData(saveData.player);
    restoreCodexData(saveData.codex);
    restoreAchievementData(saveData.achievements);
    restoreUnlocksData(saveData.unlocks);
    restoreCurrentRun(saveData.current_run);
    restoreSettings(saveData.settings);
    
    showNotification("ロードしました", "success");
    return true;
    
  } catch (error) {
    console.error("Load failed:", error);
    showNotification("ロードに失敗しました", "error");
    
    // バックアップから復元を試みる
    return loadBackup();
  }
}

// バックアップからロード
function loadBackup() {
  try {
    const backupStr = localStorage.getItem('vmo_tactics_backup');
    if (!backupStr) {
      throw new Error("No backup found");
    }
    
    const backupData = JSON.parse(backupStr);
    
    // バックアップデータで復元
    restorePlayerData(backupData.player);
    restoreCodexData(backupData.codex);
    restoreAchievementData(backupData.achievements);
    restoreUnlocksData(backupData.unlocks);
    restoreCurrentRun(backupData.current_run);
    restoreSettings(backupData.settings);
    
    showNotification("バックアップから復元しました", "warning");
    return true;
  } catch (error) {
    console.error("Backup load failed:", error);
    showNotification("バックアップからの復元も失敗しました", "error");
    return initNewGame();
  }
}
